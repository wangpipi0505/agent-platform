    $ModuleExit = $null

            Tee-Object -FilePath (Join-Path $EvidenceRoot "$Module.txt") -ErrorAction Stop

    [pscustomobject]@{module=$Module; startedAt=$StartedAt.ToString('o'); exitCode=$ModuleExit; status=$CommandStatus} |

    New-Item -ItemType Directory -Path $XmlArchive -Force | Out-Null
    Get-ChildItem -LiteralPath "$ApRoot/$Module/target/surefire-reports" -Filter 'TEST-*.xml' -ErrorAction SilentlyContinue |
        Where-Object { $_.LastWriteTime -ge $StartedAt } |
        Copy-Item -Destination $XmlArchive

2. **调用**：以下内存HTTP探针兼容5.1/7，按J1负例→J2首次写→重复写→内部读顺序运行。参数从实际资源输入，不把模板字面量当UUID。

```powershell
$OntologyBase = 'http://127.0.0.1:18182'
$ProbeSpaceId = Read-Host '本次验收空间UUID'
$ProbeReleaseId = Read-Host '该空间READY release UUID'
$ProbeAppCode = Read-Host '本次独立验收appCode'
$ProbeTenant = Read-Host '凭据绑定的platformTenant（本地开发通常local-v12）'
$ProbeEnvironment = 'production'
if (-not $env:AGENT_SECRET_ONTO_ENGINE_READ_KEY -or -not $env:AGENT_SECRET_ONTO_APP_BINDER_KEY) { throw '先准备两把独立服务账号key；不要打印值' }
$Payload = @{spaceId=$ProbeSpaceId; releaseId=$ProbeReleaseId; platformTenant=$ProbeTenant;
    appCode=$ProbeAppCode; environment=$ProbeEnvironment} | ConvertTo-Json -Compress
Add-Type -AssemblyName System.Net.Http
function Invoke-BindingProbe([string]$Method, [string]$Url, [string]$Key, [string]$JsonBody) {
    $Client = New-Object System.Net.Http.HttpClient
    $Client.Timeout = [TimeSpan]::FromSeconds(15)
    $Request = New-Object System.Net.Http.HttpRequestMessage
    $Response = $null
    try {
        if ($Method -eq 'POST') { $Request.Method = [System.Net.Http.HttpMethod]::Post }
        else { $Request.Method = [System.Net.Http.HttpMethod]::Get }
        $Request.RequestUri = [Uri]$Url
        $Request.Headers.Add('X-API-Key', $Key)
        if ($JsonBody) { $Request.Content = New-Object System.Net.Http.StringContent -ArgumentList @($JsonBody,[Text.Encoding]::UTF8,'application/json') }
        $Response = $Client.SendAsync($Request).GetAwaiter().GetResult()
        $Body = $Response.Content.ReadAsStringAsync().GetAwaiter().GetResult() | ConvertFrom-Json
        # 403等错误信封的data可能为null；兼容Environment.ps1启用的StrictMode。
        $Fact = @{httpStatus=[int]$Response.StatusCode; code=$null;
            bindingId=$null; releaseId=$null; changed=$null}
        if ($Body.PSObject.Properties['code']) { $Fact.code = $Body.code }
        $Data = $null
        if ($Body.PSObject.Properties['data']) { $Data = $Body.data }
        if ($Data) {
            foreach ($Field in @('bindingId','releaseId','changed')) {
                $Property = $Data.PSObject.Properties[$Field]
                if ($Property) { $Fact[$Field] = $Property.Value }
            }
        }
        [pscustomobject]$Fact
    } finally {
        if ($Response) { $Response.Dispose() }
        $Request.Dispose(); $Client.Dispose()
    }
}
$BindingUrl = "$OntologyBase/onto/internal/release-bindings"
$Negative = Invoke-BindingProbe 'POST' $BindingUrl $env:AGENT_SECRET_ONTO_ENGINE_READ_KEY $Payload
if ($Negative.httpStatus -ne 403 -or $Negative.code -ne 'NO_PERMISSION') { throw 'J1负例未满足' }
$First = Invoke-BindingProbe 'POST' $BindingUrl $env:AGENT_SECRET_ONTO_APP_BINDER_KEY $Payload
$Again = Invoke-BindingProbe 'POST' $BindingUrl $env:AGENT_SECRET_ONTO_APP_BINDER_KEY $Payload
$Query = 'platformTenant={0}&appCode={1}&environment={2}' -f [Uri]::EscapeDataString($ProbeTenant),[Uri]::EscapeDataString($ProbeAppCode),[Uri]::EscapeDataString($ProbeEnvironment)
$ReadBack = Invoke-BindingProbe 'GET' ($BindingUrl+'?'+$Query) $env:AGENT_SECRET_ONTO_ENGINE_READ_KEY ''
if ($First.httpStatus -ne 200 -or -not $First.changed) { throw '首次pin未创建；请使用独立新appCode，勿覆盖业务绑定' }
if ($Again.httpStatus -ne 200 -or $Again.changed -or $First.bindingId -ne $Again.bindingId) { throw '重复pin不幂等' }
if ($ReadBack.httpStatus -ne 200 -or $ReadBack.bindingId -ne $First.bindingId -or $ReadBack.releaseId -ne $ProbeReleaseId) { throw '内部读回不一致' }
# 只展示脱敏事实，不展示headers、key、带秘密的完整环境。
$Negative, $First, $Again, $ReadBack | Format-Table httpStatus,code,bindingId,releaseId,changed
```
