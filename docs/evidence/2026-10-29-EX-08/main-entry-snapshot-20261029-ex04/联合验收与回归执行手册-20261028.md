### 2.2 当前会话的定向函数

```powershell
function Test-AP([string]$Module, [string]$Classes) {
    Push-Location -LiteralPath $ApRoot
    try {
        $ArgsForTest = @('-B','-ntp','-pl',$Module,'test',
            "-Dmaven.repo.local=$ApRoot/.tools/m2",
            '-DargLine=-Xmx1024m -Djdk.net.unixdomain.tmpdir=V:/.local/tmp',
            '-Dsurefire.timeout=900','-Dsurefire.failIfNoSpecifiedTests=true')
        # 网关测试含IAM/LEGACY两种部署，ServiceTokens类加载器生命周期不可切换。
        if ($Module -eq 'platform-gateway') { $ArgsForTest += '-DreuseForks=false' }
        if ($Classes) { $ArgsForTest += "-Dtest=$Classes" }
        $CommandPreference = $ErrorActionPreference
        try {
            # 5.1原生stderr可能转成NativeCommandError；本调用按退出码判失败。
            $ErrorActionPreference = 'Continue'
            & mvn.cmd @ArgsForTest
            $TestExit = $LASTEXITCODE
        } finally { $ErrorActionPreference = $CommandPreference }
        if ($TestExit -ne 0) { throw "AP测试失败或未执行：$Module / $Classes" }
    } finally { Pop-Location }
}
# 经当前源码核实，仅这些测试依赖强制onto_test_p4的OutboxDatabase/HistoryDatabase。
$OntoP4OnlyTests = @('JdbcOutboxStoreTest','OutboxRelayTest','OutboxLeaseWaitAuditTest',
    'OutboxClockAuditTest','ReleaseEventServiceTest','TraceAuditServiceTest','ReleaseRestoreTest')
# 仅枚举类名，不执行Maven；模块全组回归可复用，不必每卡手写类清单。
function Get-ORTestClasses([string]$Module,
    [ValidateSet('BASE','P4')][string]$Database = 'BASE') {
    $Pom = [xml](Get-Content -LiteralPath "$OntoRoot/pom.xml" -Raw -Encoding UTF8)
    if ($Module -notin @($Pom.project.modules.module)) { throw '模块不在本仓reactor清单中' }
    $TestRoot = Join-Path $OntoRoot "$Module/src/test/java"
    if (-not (Test-Path -LiteralPath $TestRoot)) { return }
    $TestFiles = @(Get-ChildItem -LiteralPath $TestRoot -Filter '*Test.java' -Recurse -File)
    if (@($TestFiles | Group-Object BaseName | Where-Object { $_.Count -gt 1 }).Count -gt 0) {
        throw '存在同名测试类，先改为全限定类名清单再执行，不能合并丢项'
    }
    foreach ($File in @($TestFiles | Sort-Object BaseName)) {
        $IsP4 = $File.BaseName -in $OntoP4OnlyTests
        if (($Database -eq 'P4' -and $IsP4) -or ($Database -eq 'BASE' -and -not $IsP4)) {
            $File.BaseName
        }
    }
}
function Test-OR([string]$Module, [string]$Classes,
    [ValidateSet('BASE','P4')][string]$Database = 'BASE') {
    if ($Module -eq 'onto-release' -and [string]::IsNullOrWhiteSpace($Classes)) {
        throw 'onto-release含普通/P4两组，按2.3.1显式传测试名单与Database，不能整模块默认运行'
    }
    $SelectedTests = @($Classes -split ',' | ForEach-Object { ($_.Trim() -split '\.')[ -1 ] })
    if ($Database -eq 'BASE' -and @($SelectedTests | Where-Object { $_ -in $OntoP4OnlyTests }).Count -gt 0) {
        throw '本次名单含P4专用夹具，必须显式-Database P4'
    }
    if ($Database -eq 'P4' -and $OntoP4TestUrl -notmatch '^jdbc:postgresql://127\.0\.0\.1:15472/onto_test_p4(?:\?[^#]*)?$') {
        throw 'P4未执行：当前两夹具强制15472；先按EX-00处理端口限制，不能当测试通过'
    }
    $PreviousTestUrl = $env:ONTO_TEST_DB_URL
    Push-Location -LiteralPath $OntoRoot
    try {
        if ($Database -eq 'P4') { $env:ONTO_TEST_DB_URL = $OntoP4TestUrl }
        else { $env:ONTO_TEST_DB_URL = $OntoBaseTestUrl }
        $OntoAsciiRoot = Get-OntoAsciiRoot $OntoConfig.ONTO_LOCAL_DRIVE
        $OntoTemp = Join-Path $OntoAsciiRoot '.runtime/tmp'
        New-Item -ItemType Directory -Path $OntoTemp -Force | Out-Null
        $ArgsForTest = @('-B','-ntp','-pl',$Module,'test',
            "-DargLine=-Xmx512m -Djava.io.tmpdir=$OntoTemp -Djdk.net.unixdomain.tmpdir=$OntoTemp",
            '-Dsurefire.timeout=900','-Dsurefire.failIfNoSpecifiedTests=true')
        if ($Classes) { $ArgsForTest += "-Dtest=$Classes" }
        Invoke-OntoCommand 'mvn.cmd' $ArgsForTest
    } finally {
        [Environment]::SetEnvironmentVariable('ONTO_TEST_DB_URL', $PreviousTestUrl, 'Process')
        Pop-Location
    }
}
```

`V:/.local/tmp`沿用AP既有脚本的工程短路径；首次执行应确认本工程映射与目录存在，不接管其他工程盘符。上述函数不带 `-am`，要求2.1的当前依赖已构建。不能用 `failIfNoSpecifiedTests=false` 隐藏拼错类名。

需要重跑整个onto-release时，使用§2.3.1循环自动生成BASE/P4清单并分别归档，**不再手写两组类名**。其他卡也可用`@(Get-ORTestClasses 'onto-release' -Database BASE)`取得普通组类名，再以`-join ','`传给Test-OR；P4同理但必须显式选P4库，每组跑完立即按§2.3.1归档。助手只减少清单维护工作；卡上原定向仍必跑，不能用整组结果代替关键失败场景。新命名规则/夹具须先更新枚举与分组，空组不算通过。

### 2.3 全模块采集及放行

AP七模块依次为 `admin-common/platform-gateway/app-management/engine-runtime/skill-registry/model-gateway/monitor`；每个模块单独命令，失败也继续采集后续模块，不将它们算作通过：

```powershell
Set-Location -LiteralPath $ApRoot
$EvidenceCard = 'EX-00' # 本次实现卡改为对应EX-ID；先固定baseline，修复后使用after
$EvidencePhase = 'baseline'
$RunId = Get-Date -Format yyyyMMdd-HHmmss-fff
$EvidenceRoot = Join-Path $ApRoot "docs/evidence/$RunDay-$EvidenceCard/$EvidencePhase/$RunId"
New-Item -ItemType Directory -Path $EvidenceRoot -Force | Out-Null
foreach ($Module in @('admin-common','platform-gateway','app-management','engine-runtime','skill-registry','model-gateway','monitor')) {
    $StartedAt = Get-Date
    $ModuleExit = $null
    $CommandStatus = 'COLLECTED' # 不是通过结论
    $CommandPreference = $ErrorActionPreference
    try {
        # OR环境函数曾将当前会话设Stop；5.1下原生stderr不能让整轮提前退出。
        $ErrorActionPreference = 'Continue'
        & mvn.cmd -B -ntp -pl $Module test "-Dmaven.repo.local=$ApRoot/.tools/m2" `
            '-DargLine=-Xmx1024m -Djdk.net.unixdomain.tmpdir=V:/.local/tmp' `
            '-Dsurefire.timeout=900' '-Dmaven.test.failure.ignore=true' 2>&1 |
            Tee-Object -FilePath (Join-Path $EvidenceRoot "$Module.txt") -ErrorAction Stop
        $ModuleExit = $LASTEXITCODE
        if ($ModuleExit -ne 0) { $CommandStatus = 'COMMAND_FAILED' }
    } catch {
        $CommandStatus = 'COMMAND_FAILED'
        $_ | Out-String | Add-Content -LiteralPath (Join-Path $EvidenceRoot "$Module.txt")
    } finally { $ErrorActionPreference = $CommandPreference }
    [pscustomobject]@{module=$Module; startedAt=$StartedAt.ToString('o'); exitCode=$ModuleExit; status=$CommandStatus} |
        ConvertTo-Json | Set-Content -LiteralPath (Join-Path $EvidenceRoot "$Module-run.json") -Encoding utf8
    $XmlArchive = Join-Path $EvidenceRoot "$Module-xml"
    New-Item -ItemType Directory -Path $XmlArchive -Force | Out-Null
    Get-ChildItem -LiteralPath "$ApRoot/$Module/target/surefire-reports" -Filter 'TEST-*.xml' -ErrorAction SilentlyContinue |
        Where-Object { $_.LastWriteTime -ge $StartedAt } |
        Copy-Item -Destination $XmlArchive
}
```

                Tee-Object -FilePath (Join-Path $GroupDir 'raw.txt')

                startedAt=$StartedAt.ToString('o'); status=$RunStatus} | ConvertTo-Json -Depth 4 |
                Set-Content -LiteralPath (Join-Path $GroupDir 'run.json') -Encoding utf8
        }
    }
}

### 3.1 最小绑定协议冒烟（三步，可早于完整EX-07）

这条短路径不依赖EX-03/04/05/06的导入/生成脚本。先完成EX-00基线、EX-01服务配置与EX-02请求字段修复；OR已部署且手工用现有管理入口创建独立验收空间、READY release、readonly/binder两服务账号（不同platformActor，binder显式授release.bind）。如果没有这些真实输入，先按已授权范围准备再验证，不编造通过；文档示例本身不产生资源，实际资源/凭据须按对应EX证据追溯。

1. **准备**：把真实spaceId/releaseId/appCode/environment和已验platformTenant记入本次证据；同一进程中仅按权限配置`AGENT_SECRET_ONTO_ENGINE_READ_KEY`/`AGENT_SECRET_ONTO_APP_BINDER_KEY`，不要输出它们。无平台应用时appCode可用独立协议探针编码，此时更不能算J9。要验证平台真实发布，须另有有效候选及正常评测/发布前提。
2. **调用**：以下内存HTTP探针兼容5.1/7，按J1负例→J2首次写→重复写→内部读顺序运行。参数从实际资源输入，不把模板字面量当UUID。

```powershell
$OntologyBase = 'http://127.0.0.1:18182'
$ProbeSpaceId = Read-Host '本次验收空间UUID'
$ProbeReleaseId = Read-Host '该空间READY release UUID'
$ProbeAppCode = Read-Host '本次独立验收appCode'
$ProbeTenant = Read-Host '凭据绑定的platformTenant（本地开发通常local-v12）'
$ProbeEnvironment = 'production'
if (-not $env:AGENT_SECRET_ONTO_ENGINE_READ_KEY -or -not $env:AGENT_SECRET_ONTO_APP_BINDER_KEY) { throw '先准备两把独立服务账号key；不要打印值' }
$Payload = @{spaceId=$ProbeSpaceId; releaseId=$ProbeReleaseId; platformTenant=$ProbeTenant;
    appCode=$ProbeAppCode; environment=$ProbeEnvironment} | ConvertTo-Json -Compress
Add-Type -AssemblyName System.Net.Http
function Invoke-BindingProbe([string]$Method, [string]$Url, [string]$Key, [string]$JsonBody) {
    $Client = New-Object System.Net.Http.HttpClient
    $Client.Timeout = [TimeSpan]::FromSeconds(15)
    $Request = New-Object System.Net.Http.HttpRequestMessage
    $Response = $null
    try {
        if ($Method -eq 'POST') { $Request.Method = [System.Net.Http.HttpMethod]::Post }
        else { $Request.Method = [System.Net.Http.HttpMethod]::Get }
        $Request.RequestUri = [Uri]$Url
        $Request.Headers.Add('X-API-Key', $Key)
        if ($JsonBody) { $Request.Content = New-Object System.Net.Http.StringContent -ArgumentList @($JsonBody,[Text.Encoding]::UTF8,'application/json') }
        $Response = $Client.SendAsync($Request).GetAwaiter().GetResult()
        $Body = $Response.Content.ReadAsStringAsync().GetAwaiter().GetResult() | ConvertFrom-Json
        # 403等错误信封的data可能为null；兼容Environment.ps1启用的StrictMode。
        $Fact = @{httpStatus=[int]$Response.StatusCode; code=$null;
            bindingId=$null; releaseId=$null; changed=$null}
        if ($Body.PSObject.Properties['code']) { $Fact.code = $Body.code }
        $Data = $null
        if ($Body.PSObject.Properties['data']) { $Data = $Body.data }
        if ($Data) {
            foreach ($Field in @('bindingId','releaseId','changed')) {
                $Property = $Data.PSObject.Properties[$Field]
                if ($Property) { $Fact[$Field] = $Property.Value }
            }
        }
        [pscustomobject]$Fact
    } finally {
        if ($Response) { $Response.Dispose() }
        $Request.Dispose(); $Client.Dispose()
    }
}
$BindingUrl = "$OntologyBase/onto/internal/release-bindings"
$Negative = Invoke-BindingProbe 'POST' $BindingUrl $env:AGENT_SECRET_ONTO_ENGINE_READ_KEY $Payload
if ($Negative.httpStatus -ne 403 -or $Negative.code -ne 'NO_PERMISSION') { throw 'J1负例未满足' }
$First = Invoke-BindingProbe 'POST' $BindingUrl $env:AGENT_SECRET_ONTO_APP_BINDER_KEY $Payload
$Again = Invoke-BindingProbe 'POST' $BindingUrl $env:AGENT_SECRET_ONTO_APP_BINDER_KEY $Payload
$Query = 'platformTenant={0}&appCode={1}&environment={2}' -f [Uri]::EscapeDataString($ProbeTenant),[Uri]::EscapeDataString($ProbeAppCode),[Uri]::EscapeDataString($ProbeEnvironment)
$ReadBack = Invoke-BindingProbe 'GET' ($BindingUrl+'?'+$Query) $env:AGENT_SECRET_ONTO_ENGINE_READ_KEY ''
if ($First.httpStatus -ne 200 -or -not $First.changed) { throw '首次pin未创建；请使用独立新appCode，勿覆盖业务绑定' }
if ($Again.httpStatus -ne 200 -or $Again.changed -or $First.bindingId -ne $Again.bindingId) { throw '重复pin不幂等' }
if ($ReadBack.httpStatus -ne 200 -or $ReadBack.bindingId -ne $First.bindingId -or $ReadBack.releaseId -ne $ProbeReleaseId) { throw '内部读回不一致' }
# 只展示脱敏事实，不展示headers、key、带秘密的完整环境。
$Negative, $First, $Again, $ReadBack | Format-Table httpStatus,code,bindingId,releaseId,changed
```
